1. Calculate the sum of all elements in an array

public class Main {
    public static void main(String[] args) {
        int[] arr = {10, 20, 30, 40, 50};
        int sum = 0;

        for (int i = 0; i < arr.length; i++) {
            sum += arr[i];
        }

        System.out.println("Sum = " + sum);
    }
}
Output:



Sum = 150


2. Calculate the average of all elements in an array



public class Main {
    public static void main(String[] args) {
        int[] arr = {10, 20, 30, 40, 50};
        int sum = 0;

        for (int i = 0; i < arr.length; i++) {
            sum += arr[i];
        }

        double average = (double) sum / arr.length;
        System.out.println("Average = " + average);
    }
}
Output:



Average = 30.0

5. Count even and odd numbers in an array

public class Main {
    public static void main(String[] args) {
        int[] arr = {10, 15, 20, 25, 30, 35};
        int even = 0, odd = 0;

        for (int i = 0; i < arr.length; i++) {
            if (arr[i] % 2 == 0) {
                even++;
            } else {
                odd++;
            }
        }

        System.out.println("Even count = " + even);
        System.out.println("Odd count = " + odd);
    }
}
Output:




Even count = 3
Odd count = 3

6. Search for an element using linear search



public class Main {
    public static void main(String[] args) {
        int[] arr = {10, 20, 30, 40, 50};
        int search = 30;
        int index = -1;

        for (int i = 0; i < arr.length; i++) {
            if (arr[i] == search) {
                index = i;
                break;
            }
        }

        if (index != -1) {
            System.out.println("Element found at index " + index);
        } else {
            System.out.println("Element not found");
        }
    }
}
Output:



Element found at index 2

7. Count positive numbers, negative numbers, and zeros



public class Main {
    public static void main(String[] args) {
        int[] arr = {10, -5, 0, 20, -15, 0, 30};
        int positive = 0, negative = 0, zeros = 0;

        for (int i = 0; i < arr.length; i++) {
            if (arr[i] > 0) {
                positive++;
            } else if (arr[i] < 0) {
                negative++;
            } else {
                zeros++;
            }
        }

        System.out.println("Positive = " + positive);
        System.out.println("Negative = " + negative);
        System.out.println("Zeros = " + zeros);
    }
}
Output:


Positive = 3
Negative = 2
Zeros = 2  

8. Print array elements from last to first


public class Main {
    public static void main(String[] args) {
        int[] arr = {10, 20, 30, 40, 50};

        for (int i = arr.length - 1; i >= 0; i--) {
            System.out.print(arr[i] + " ");
        }
    }
}
Output:



50 40 30 20 10
